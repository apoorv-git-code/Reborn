# This file is both pip-installable and a complete local setup guide.

#
# 1. Install Python 3.11 on macOS (Homebrew):
#      brew install python@3.11
#
# 2. Create a Python 3.11 virtual environment from the project folder:
#      "$(brew --prefix python@3.11)/bin/python3.11" -m venv .venv311
#
# 3. Activate it:
#      source .venv311/bin/activate
#
# 4. Confirm the version:
#      python --version
#    Expected: Python 3.11.x
#
# 5. Upgrade pip and install all dependencies:
#      python -m pip install --upgrade pip
#      python -m pip install -r code/requirements.txt
#
# 6. Run from a directory containing train.csv, dev.csv,
#    dev_winners.csv and test.csv:
#      python code/main.py
#
# 7. Validate the generated shortlist (when check_format.py is available):
#      python check_format.py submission.csv test.csv
